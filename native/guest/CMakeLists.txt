cmake_minimum_required(VERSION 3.22)
project(superman_guest_compile_check LANGUAGES CXX)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Use Android NDK arm64-v8a")
endif()
set(SR_RECOMP_ROOT "" CACHE PATH "Local superman_returns_recomp project")
set(SR_GUEST_HEADERS "${CMAKE_CURRENT_LIST_DIR}/../../.tools/guest-sdk/include" CACHE PATH "Private Android SDK header overlay")
if(NOT EXISTS "${SR_RECOMP_ROOT}/port/generated/default/sources.cmake" OR NOT EXISTS "${SR_GUEST_HEADERS}/rex/ppc.h")
    message(FATAL_ERROR "Run tools/prepare_guest.py --recomp <path> first and set SR_RECOMP_ROOT")
endif()
include("${SR_RECOMP_ROOT}/port/generated/default/sources.cmake")
# An archive proves compilation only. Runtime symbols remain unresolved by design.
add_library(superman_guest_codegen STATIC ${GENERATED_SOURCES})
target_compile_features(superman_guest_codegen PRIVATE cxx_std_23)
target_include_directories(superman_guest_codegen PRIVATE "${SR_GUEST_HEADERS}" "${SR_RECOMP_ROOT}/port/generated/default")
target_compile_definitions(superman_guest_codegen PRIVATE SPDLOG_FMT_EXTERNAL SIMDE_ENABLE_NATIVE_ALIASES)
target_compile_options(superman_guest_codegen PRIVATE -fno-strict-aliasing -fwrapv -Wno-unused-variable -Wno-unused-label)
target_precompile_headers(superman_guest_codegen PRIVATE "${SR_RECOMP_ROOT}/port/generated/default/superman_returns_pch.h")
